/* ═══════════════════════ OD PLANNER ═══════════════════════ * * CARA PASANG (2 langkah): * * 1. Di fungsi doPost, tambahkan 2 baris ini di dalam switch, * tepat di bawah baris case 'deleteOrgStructure': ... * * case 'saveODTask': return saveODTask_(data); * case 'deleteODTask': return deleteODTask_(data); * * 2. Tempel seluruh isi di bawah ini di PALING BAWAH Code.gs. * * Lalu: Deploy → Manage deployments → ikon pensil → Version: New version → Deploy * Sheet "OD Planner" dibuat otomatis saat task pertama disimpan. */ var PLANNER_SHEET = 'OD Planner'; var PLANNER_HEAD = ['id','task','pic','kategori','deadline','status','prioritas','catatan','created_at','updated_at']; function plannerSheet_() { var sh = getOrCreateSheet_(PLANNER_SHEET, PLANNER_HEAD); // Simpan semua kolom sebagai teks supaya tanggal tidak diubah formatnya oleh Sheets sh.getRange(1, 1, sh.getMaxRows(), PLANNER_HEAD.length).setNumberFormat('@'); return sh; } function plannerNow_() { return Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm'); } function saveODTask_(data) { var lock = LockService.getScriptLock(); try { lock.waitLock(30000); var t = data.task || {}; var task = String(t.task || '').trim(); if (!task) return json_({ success: false, error: 'Nama task kosong' }); var sh = plannerSheet_(); var all = sh.getDataRange().getValues(); var id = String(t.id || '').trim() || ('T' + new Date().getTime()); var now = plannerNow_(); var rowIdx = -1, created = now; for (var i = 1; i < all.length; i++) { if (String(all[i][0]).trim() === id) { rowIdx = i + 1; created = all[i][8] || now; break; } } var row = [ id, task, String(t.pic || '').trim(), String(t.kategori || '').trim(), String(t.deadline || '').trim(), String(t.status || 'To Do').trim(), String(t.prioritas || 'Medium').trim(), String(t.catatan || '').trim(), created, now ]; if (rowIdx > 0) { sh.getRange(rowIdx, 1, 1, PLANNER_HEAD.length).setValues([row]); } else { sh.getRange(sh.getLastRow() + 1, 1, 1, PLANNER_HEAD.length).setValues([row]); } return json_({ success: true, id: id, updated: rowIdx > 0 }); } catch (err) { return json_({ success: false, error: err.message }); } finally { lock.releaseLock(); } } function deleteODTask_(data) { var lock = LockService.getScriptLock(); try { lock.waitLock(30000); var id = String(data.id || '').trim(); if (!id) return json_({ success: false, error: 'ID task kosong' }); var sh = ss_().getSheetByName(PLANNER_SHEET); if (!sh) return json_({ success: true, deleted: 0 }); var all = sh.getDataRange().getValues(); for (var i = all.length - 1; i >= 1; i--) { if (String(all[i][0]).trim() === id) { sh.deleteRow(i + 1); return json_({ success: true, deleted: 1 }); } } return json_({ success: true, deleted: 0 }); } catch (err) { return json_({ success: false, error: err.message }); } finally { lock.releaseLock(); } }
